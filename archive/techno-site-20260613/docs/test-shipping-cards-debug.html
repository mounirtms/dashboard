🔍 Shipping Cards Debug Test

Test Instructions

  1. Open the checkout page: https://dev.technostationery.com/checkout/#shipping
  2. Open browser Developer Console (F12)
  3. Copy and paste the debug script below into the console
  4. Select a wilaya (e.g., Biskra, Blida, or Alger)
  5. Watch the console for debug output

📋 Debug Script (Copy to Console)

// ==== SHIPPING CARDS DEBUG SCRIPT ====
(function() {
    console.log('%c🔍 SHIPPING CARDS DEBUG STARTED', 'background: #4CAF50; color: white; padding: 5px 10px; font-weight: bold;');
    
    // Function to check component
    function checkShippingCards() {
        // Get UI Registry
        var registry = require.s.contexts._.defined['uiRegistry'];
        
        if (!registry) {
            console.error('❌ UI Registry not found');
            return;
        }
        
        console.log('✅ UI Registry found');
        
        // Find shipping cards component
        var component = null;
        registry.get('checkout.steps.shipping-step.shippingAddress.before-shipping-method-form.shipping-method-cards', function(comp) {
            component = comp;
        });
        
        if (!component) {
            console.error('❌ Shipping cards component not found');
            console.log('💡 Available components:', Object.keys(registry.get()));
            return;
        }
        
        console.log('✅ Shipping cards component found');
        console.log('📊 Component state:', {
            visible: component.isVisible(),
            loading: component.isLoading(),
            methods: component.shippingMethods().length,
            selected: component.selectedMethod(),
            region: component.currentRegion(),
            error: component.errorMessage()
        });
        
        // Check shipping service
        var shippingService = require.s.contexts._.defined['Magento_Checkout/js/model/shipping-service'];
        if (shippingService) {
            var rates = shippingService.getShippingRates()();
            console.log('📦 Shipping service rates:', rates);
            console.log('📦 Number of rates:', rates ? rates.length : 0);
            
            if (rates && rates.length > 0) {
                console.log('🔍 Inspecting each rate:');
                rates.forEach(function(rate, idx) {
                    console.log('   Rate #' + idx + ':', {
                        carrier: rate.carrier_code,
                        method: rate.method_code,
                        available: rate.available,
                        amount: rate.amount,
                        title: rate.method_title
                    });
                });
            }
        }
        
        // Check wrapper element
        var wrapper = document.querySelector('.shipping-methods-cards-wrapper');
        if (wrapper) {
            var styles = window.getComputedStyle(wrapper);
            console.log('📐 Wrapper element styles:', {
                display: styles.display,
                visibility: styles.visibility,
                opacity: styles.opacity,
                height: styles.height,
                overflow: styles.overflow
            });
        } else {
            console.error('❌ Wrapper element not found in DOM');
        }
        
        // Check quote
        var quote = require.s.contexts._.defined['Magento_Checkout/js/model/quote'];
        if (quote) {
            var address = quote.shippingAddress();
            console.log('📍 Quote shipping address:', {
                regionId: address ? address.regionId : 'N/A',
                region: address ? address.region : 'N/A',
                regionCode: address ? address.regionCode : 'N/A',
                countryId: address ? address.countryId : 'N/A'
            });
            
            var method = quote.shippingMethod();
            console.log('🚚 Quote shipping method:', method);
        }
    }
    
    // Run initial check
    console.log('⏳ Running initial check...');
    checkShippingCards();
    
    // Make function globally available
    window.debugShippingCards = checkShippingCards;
    
    console.log('%c✅ Debug script loaded successfully', 'background: #4CAF50; color: white; padding: 5px 10px;');
    console.log('💡 Run window.debugShippingCards() anytime to check component state');
    console.log('💡 Or just call: debugShippingCards()');
    
})();

⚠️ If Rates Show method_code: null

This indicates a backend issue. Check:

  1. Mageplaza TableRate module configuration
  2. Method records in database table mageplaza_tablerate_method
  3. Rate records in mageplaza_tablerate_rate
  4. Module enabled in Stores > Configuration > Sales > Shipping Methods
